set(EXAMPLES_DIR ${CMAKE_CURRENT_LIST_DIR})
file(GLOB children RELATIVE ${EXAMPLES_DIR} ${EXAMPLES_DIR}/*)

foreach(child ${children})
    set(child_dir "${EXAMPLES_DIR}/${child}")

    if("${child}" STREQUAL "configs")
        continue()
    endif()

    if(IS_DIRECTORY ${child_dir})
        set(kconfig_path "${child_dir}/kconfig")
        if(EXISTS ${kconfig_path})
            file(STRINGS ${kconfig_path} kconfig_lines REGEX "^config ")
            if(kconfig_lines)
                string(REGEX MATCH "config[ \t]+([A-Za-z0-9_]+)" _ "${kconfig_lines}")
                set(config_symbol "${CMAKE_MATCH_1}")
                set(config_cmake_var "CONFIG_${config_symbol}")
                if(DEFINED ${config_cmake_var})
                    if(${${config_cmake_var}})
                        message(STATUS "Including ${child} because ${config_cmake_var} is enabled")
                        add_subdirectory(${child_dir})
                    endif()
                endif()
            else()
                message(WARNING "No 'config' entry found in ${kconfig_path}")
            endif()
        else()
            message(WARNING "No Kconfig file in ${child_dir}, skipping")
        endif()
    endif()
endforeach()
